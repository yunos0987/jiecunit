{#include <jiecunit.txt>}

// Test internal validation functions directly
TEST_S(test_jiectest_validate_gt)
	var
		jt_int32: JiecTest_t;
		jt_uint32: JiecTest_t;
		jt_double: JiecTest_t;
		jt_string: JiecTest_t;
		jt_time: JiecTest_t;
		jt_int64: JiecTest_t;
		jt_uint64: JiecTest_t;
	end
	{st}
	JiecTest_initialize(jt_int32);
	EXPECT_TRUE(JiecTest_validateGT_INT32(jt_int32, 2, 1, 'INT32: 2 > 1', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_INT32(jt_int32, 2, 2, 'INT32: 2 > 2', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_INT32(jt_int32, 1, 2, 'INT32: 1 > 2', false, TestName, __LINE__));

	JiecTest_initialize(jt_uint32);
	EXPECT_TRUE(JiecTest_validateGT_UINT32(jt_uint32, 2, 1, 'UINT32: 2 > 1', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_UINT32(jt_uint32, 2, 2, 'UINT32: 2 > 2', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_UINT32(jt_uint32, 1, 2, 'UINT32: 1 > 2', false, TestName, __LINE__));

	JiecTest_initialize(jt_double);
	EXPECT_TRUE(JiecTest_validateGT_DOUBLE(jt_double, 2.0, 1.0, 'DOUBLE: 2.0 > 1.0', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_DOUBLE(jt_double, 2.0, 2.0, 'DOUBLE: 2.0 > 2.0', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_DOUBLE(jt_double, 1.0, 2.0, 'DOUBLE: 1.0 > 2.0', false, TestName, __LINE__));

	JiecTest_initialize(jt_string);
	EXPECT_TRUE(JiecTest_validateGT_STRING(jt_string, 'B', 'A', 'STRING: "B" > "A"', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_STRING(jt_string, 'A', 'A', 'STRING: "A" > "A"', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_STRING(jt_string, 'A', 'B', 'STRING: "A" > "B"', false, TestName, __LINE__));

{#if not (_KEYENCE or _MITSUBISHI)}
	JiecTest_initialize(jt_time);
	EXPECT_TRUE(JiecTest_validateGT_TIME(jt_time, time#2s, time#1s, 'TIME: 2s > 1s', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_TIME(jt_time, time#2s, time#2s, 'TIME: 2s > 2s', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_TIME(jt_time, time#1s, time#2s, 'TIME: 1s > 2s', false, TestName, __LINE__));

	JiecTest_initialize(jt_time);
	EXPECT_TRUE(JiecTest_validateGT_DATE(jt_time, date#2026-03-19, date#2026-03-18, 'DATE: 2026-03-19 > 2026-03-18', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_DATE(jt_time, date#2026-03-18, date#2026-03-18, 'DATE: equal dates', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_DATE(jt_time, date#2026-03-18, date#2026-03-19, 'DATE: 2026-03-18 > 2026-03-19', false, TestName, __LINE__));

	JiecTest_initialize(jt_time);
	EXPECT_TRUE(JiecTest_validateGT_TOD(jt_time, tod#12:34:57, tod#12:34:56, 'TOD: 12:34:57 > 12:34:56', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_TOD(jt_time, tod#12:34:56, tod#12:34:56, 'TOD: equal times', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_TOD(jt_time, tod#12:34:56, tod#12:34:57, 'TOD: 12:34:56 > 12:34:57', false, TestName, __LINE__));

	JiecTest_initialize(jt_time);
	EXPECT_TRUE(JiecTest_validateGT_DT(jt_time, dt#2026-03-18-12:34:57, dt#2026-03-18-12:34:56, 'DT: 12:34:57 > 12:34:56', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_DT(jt_time, dt#2026-03-18-12:34:56, dt#2026-03-18-12:34:56, 'DT: equal timestamps', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_DT(jt_time, dt#2026-03-18-12:34:56, dt#2026-03-18-12:34:57, 'DT: 12:34:56 > 12:34:57', false, TestName, __LINE__));

	JiecTest_initialize(jt_int64);
	EXPECT_TRUE(JiecTest_validateGT_INT64(jt_int64, 2, 1, 'INT64: 2 > 1', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_INT64(jt_int64, 2, 2, 'INT64: 2 > 2', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_INT64(jt_int64, 1, 2, 'INT64: 1 > 2', false, TestName, __LINE__));

	JiecTest_initialize(jt_uint64);
	EXPECT_TRUE(JiecTest_validateGT_UINT64(jt_uint64, 2, 1, 'UINT64: 2 > 1', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_UINT64(jt_uint64, 2, 2, 'UINT64: 2 > 2', false, TestName, __LINE__));
	EXPECT_FALSE(JiecTest_validateGT_UINT64(jt_uint64, 1, 2, 'UINT64: 1 > 2', false, TestName, __LINE__));
{#endif}
	{end}
END_TEST_S

// Test EXPECT_GT and ASSERT_GT macros with all data types
TEST_S(test_gt_macro)
	{st}
	EXPECT_GT_INT(2, 1);
	EXPECT_GT_INT(0, -1);
	EXPECT_GT_UINT(1, 0);
	EXPECT_GT_DOUBLE(0.0, -1.0);
	EXPECT_GT_STRING('B', 'A');
{#if not (_KEYENCE or _MITSUBISHI)}
	EXPECT_GT_TIME(time#2s, time#1s);
	EXPECT_GT_DATE(date#2026-03-19, date#2026-03-18);
	EXPECT_GT_TOD(tod#12:34:57, tod#12:34:56);
	EXPECT_GT_DT(dt#2026-03-18-12:34:57, dt#2026-03-18-12:34:56);
	ASSERT_GT_DATE(date#2026-03-19, date#2026-03-18);
	ASSERT_GT_TOD(tod#12:34:57, tod#12:34:56);
	ASSERT_GT_DT(dt#2026-03-18-12:34:57, dt#2026-03-18-12:34:56);
{#endif}
	// ...他データ型・メッセージ・ASSERT_GTも同様に追加...
	{end}
END_TEST_S
