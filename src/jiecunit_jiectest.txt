type
	(*{doc The structure to manage the test cases and test results.}*)
	JiecTest_t: struct
		(*{doc The number of registered test cases.}*)
		num_of_tests: dint;
		(*{doc The names of registered test cases.}*)
		testnames: array[MAX_TESTS] of string[TESTNAME_MAX_SIZE];
		//
		tmp_cycle: dint; // There is a active test, if tmp_cycle >= 0.
		tmp_active_test_index: dint;
		tmp_is_test_available: bool; // error, if the test specified test is not found.
		(*{doc The number of failed assertions in the active test.}*)
		num_of_failures: dint;
	end
end

(*{doc Initialize the JiecTest_t structure.
Sets all fields including test case count, test names, cycle counter,
and failure count to their initial states.
@inout self The JiecTest_t structure to initialize.
@return Always returns true.}*)
function JiecTest_initialize: bool
	_PROLOGUE_OF_FUNCTION_()
	var_in_out self: JiecTest_t; end
	var_temp
		test_index: dint;
	end
	{st}
	self.num_of_tests := 0;
	for test_index := 0 to MAX_TESTS - 1 do
		self.testnames[test_index] := '';
	end_for;

	self.tmp_cycle := dint#__DINT_MIN__;
	self.tmp_active_test_index := -1;
	self.tmp_is_test_available := false;
	self.num_of_failures := 0;
	{end}
end

(*{doc Register a test case.
Returns -1 if the test name is already registered.
@inout self The JiecTest_t structure.
@input test_name The name of the test case to register.
@return The total number of registered test cases, or -1 if already registered.}*)
function JiecTest_register: dint
	_PROLOGUE_OF_FUNCTION_()
	var_in_out self: JiecTest_t; end
	var_input
		(*{doc The name of the test case to register.}*)
		test_name: string[TESTNAME_MAX_SIZE];
	end
	var_temp
		test_index: dint;
	end
	{st}
	// ignore, if it is already registered.
	for test_index := 0 to self.num_of_tests - 1 do
		if test_name = self.testnames[test_index] then
			JiecTest_register := -1;
			return;
		end_if;
	end_for;

	self.testnames[self.num_of_tests] := test_name;
	inc(self.num_of_tests);
	JiecTest_register := self.num_of_tests;
	{end}
end

(*{doc Return bool value indicating whether there is no registered test case.
@inout self The JiecTest_t structure.
@return True if there are no registered test cases, false otherwise.}*)
function JiecTest_empty: bool
	_PROLOGUE_OF_FUNCTION_()
	var_in_out self: JiecTest_t; end
	{st}
	JiecTest_empty := self.num_of_tests = 0;
	{end}
end

(*{doc Report a test failure.
Outputs failure information to the console and increments the failure counter.
@inout self The JiecTest_t structure.
@input message The failure message (optional).
@input die Whether to abort the test on failure. Default: true
@input file The file name where the failure occurred.
@input line The line number where the failure occurred.
@return Always returns true.}*)
function JiecTest_failure: bool
	_PROLOGUE_OF_FUNCTION_()
	_DECLARE_CONSOLE_IN_POU_()
	var_in_out self: JiecTest_t; end
	var_input
		(*{doc The failure message (optional).}*)
		message: string[STRING_MAX_SIZE] := '';
		(*{doc Whether to abort the test on failure. Default: true}*)
		die: bool := true;
		(*{doc The file name where the failure occurred.}*)
		file: string[FILENAME_MAX_SIZE];
		(*{doc The line number where the failure occurred.}*)
		line: dint;
	end
	var_temp
		msg: string[STRING_MAX_SIZE];
	end
	{st}
	msg := jiec_concat(file, ':', dint_to_string(line), '@', dint_to_string(JiecTest_currentCycle(self)));
	if message <> '' then
		msg := jiec_concat(msg, ' ', message);
	end_if;
	puts(msg);

	inc(self.num_of_failures);
	{end}
end

(*{doc Get a bool value indicating whether there is an active test case.
@inout self The JiecTest_t structure.
@return True if there is an active test case, false otherwise.}*)
function JiecTest_isActive: bool
	_PROLOGUE_OF_FUNCTION_()
	var_in_out self: JiecTest_t; end
	{st}
	JiecTest_isActive := self.tmp_cycle >= 0;
	{end}
end

(*{doc Get the index of the active test case.
@inout self The JiecTest_t structure.
@return The index of the active test case.}*)
function JiecTest_activeTestIndex: dint
	_PROLOGUE_OF_FUNCTION_()
	var_in_out self: JiecTest_t; end
	{st}
	JiecTest_activeTestIndex := self.tmp_active_test_index;
	{end}
end

(*{doc Get the name of the active test case.
@inout self The JiecTest_t structure.
@return The name of the active test case.}*)
function JiecTest_activeTestname: string[TESTNAME_MAX_SIZE]
	_PROLOGUE_OF_FUNCTION_()
	var_in_out self: JiecTest_t; end
	{st}
	JiecTest_activeTestname := self.testnames[self.tmp_active_test_index];
	{end}
end

(*{doc Activate a test case and set the cycle counter to 0.
@inout self The JiecTest_t structure.
@return Always returns true.}*)
function JiecTest_activate: bool
	_PROLOGUE_OF_FUNCTION_()
	var_in_out self: JiecTest_t; end
	{st}
	self.tmp_cycle := 0;
	{end}
end

(*{doc Deactivate the current test case.
@inout self The JiecTest_t structure.
@return Always returns true.}*)
function JiecTest_deactivate: bool
	_PROLOGUE_OF_FUNCTION_()
	var_in_out self: JiecTest_t; end
	{st}
	self.tmp_cycle := dint#__DINT_MIN__;
	{end}
end

(*{doc Get the current cycle number of the active test case.
@inout self The JiecTest_t structure.
@return The current cycle number.}*)
function JiecTest_currentCycle: dint
	_PROLOGUE_OF_FUNCTION_()
	var_in_out self: JiecTest_t; end
	{st}
	JiecTest_currentCycle := self.tmp_cycle;
	{end}
end

(*{doc Increment the cycle counter for the active test case.
@inout self The JiecTest_t structure.
@return False if overflow occurred, true otherwise.}*)
function JiecTest_nextCycle: bool
	_PROLOGUE_OF_FUNCTION_()
	var_in_out self: JiecTest_t; end
	{st}
	JiecTest_nextCycle := self.tmp_cycle <> dint#__DINT_MAX__; // false, if overflow happend.
	self.tmp_cycle := self.tmp_cycle + 1;
	{end}
end

(*{doc Get the test name at the specified index.
@inout self The JiecTest_t structure.
@input test_index The index of the test case.
@return The test name, or an error message if the index is out of range.}*)
function JiecTest_testname: string[TESTNAME_MAX_SIZE]
	_PROLOGUE_OF_FUNCTION_()
	var_in_out self: JiecTest_t; end
	var_input
		(*{doc The index of the test case.}*)
		test_index: dint;
	end
	{st}
	if (0 <= test_index) and (test_index < self.num_of_tests) then
		JiecTest_testname := self.testnames[test_index];
	else
		JiecTest_testname := '!No available tests!';
	end_if;
	{end}
end

{#define DEFINE_JiecTest_validateEQ_DATATYPE(DATATYPE, datatype_, to_string)
	(*${doc Validate that the expected value equals the actual value.$n
	@inout self The JiecTest_t structure.$n
	@input expected The expected value.$n
	@input actual The actual value.$n
	@input message The failure message.$n
	@input die Whether to abort the test on failure.$n
	@input file The source file name.$n
	@input line The line number.$n
	@return True if the values are equal, false otherwise.$}*)
	function JiecTest_validateEQ_ @@ DATATYPE: bool
	_PROLOGUE_OF_FUNCTION_()
	_DECLARE_CONSOLE_IN_POU_()
	var_in_out self: JiecTest_t; end
	var_input
		(*${doc The expected value.$}*)
		expected: datatype_;
		(*${doc The actual value.$}*)
		actual: datatype_;
		(*${doc The failure message.$}*)
		message: string[128];

		file: string[FILENAME_MAX_SIZE];

		line: dint;
	end

		return;
	end_if;

	puts(jiec_concat('Expected: ', to_string(expected)));
	puts(jiec_concat('Actual: ', to_string(actual)));

	JiecTest_failure(self:=self, message:=message, die:=die, file:=file, line:=line);
	${end$}
	end}

{#define DEFINE_JiecTest_validateCMP_DATATYPE(CMP, DATATYPE, operator_, datatype_, to_string)
	(*${doc Validate a comparison between two values using the specified operator.$n
@inout self The JiecTest_t structure.$n

@input rval The right-hand side value.$n
@input message The failure message.$n
@input die Whether to abort the test on failure.$n
@input file The source file name.$n
@input line The line number.$n
@return True if the comparison holds, false otherwise.$}*)
	function JiecTest_validate @@ CMP @@ _ @@ DATATYPE: bool
	_PROLOGUE_OF_FUNCTION_()
	_DECLARE_CONSOLE_IN_POU_()
	var_in_out self: JiecTest_t; end
	var_input
		(*${doc The left-hand side value.$}*)
		lval: datatype_;
		(*${doc The right-hand side value.$}*)
		rval: datatype_;
		(*${doc The failure message.$}*)

		(*${doc The source file name.$}*)
		file: string[FILENAME_MAX_SIZE];
		(*${doc The line number.$}*)
		line: dint;
	end
	${st$}
	JiecTest_validate @@ CMP @@ _ @@ DATATYPE := lval operator_ rval;
	if JiecTest_validate @@ CMP @@ _ @@ DATATYPE then
		return;
	end_if;

	puts(jiec_concat('Expected: ', to_string(lval), @operator_, to_string(rval)));
	puts(jiec_concat('Actual: ', to_string(lval), ' vs ', to_string(rval)));

	JiecTest_failure(self:=self, message:=message, die:=die, file:=file, line:=line);
	${end$}
	end
}

(*{doc Validate that two floating-point values are nearly equal within the specified absolute error tolerance.
@inout self The JiecTest_t structure.
@input val1 The first value.
@input val2 The second value.
@input abs_error The allowed absolute error.
@input message The failure message.
@input die Whether to abort the test on failure. Default: true
@input file The source file name.
@input line The line number.
@return True if the values are equal within the tolerance, false otherwise.}*)
function JiecTest_validateNEAR: bool
_PROLOGUE_OF_FUNCTION_()
_DECLARE_CONSOLE_IN_POU_()
var_in_out self: JiecTest_t; end
var_input
	(*{doc The first value.}*)
	val1: lreal;
	(*{doc The second value.}*)
	val2: lreal;
	(*{doc The allowed absolute error.}*)
	abs_error: lreal;
	(*{doc The failure message.}*)
	message: string[128];
	(*{doc Whether to abort the test on failure.}*)
	die: bool := true;
	(*{doc The source file name.}*)
	file: string[FILENAME_MAX_SIZE];
	(*{doc The line number.}*)
	line: dint;
end
{st}
if abs_error < 0.0 then
	JiecTest_failure(self:=self, message:='Absolute error cannot be negative.', die:=die, file:=file, line:=line);
	JiecTest_validateNEAR := false;
	return;
end_if;

JiecTest_validateNEAR := abs(val1 - val2) <= abs_error;
if JiecTest_validateNEAR then
	return;
end_if;

puts(jiec_concat('Value1: ', lreal_to_string(val1)));
puts(jiec_concat('Value2: ', lreal_to_string(val2)));
puts(jiec_concat('Value1 - Value2: ', lreal_to_string(val1 - val2)));

JiecTest_failure(self:=self, message:=message, die:=die, file:=file, line:=line);
{end}
end
